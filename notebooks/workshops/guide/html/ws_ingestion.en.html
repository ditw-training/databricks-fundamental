        # Step 12 — left join orders with customers
orders_enriched = (
    orders_df
    .join(customers_df, on="customer_id", how="left")
    .select("order_id", "customer_id", "first_name", "last_name",
            "country", "total_amount", "payment_method")
)

# Orders and revenue per country
orders_by_country = (
    orders_enriched.groupBy("country")
    .agg(F.count("*").alias("order_count"), F.sum("total_amount").alias("total_revenue"))
    .orderBy(F.desc("total_revenue"))
)
